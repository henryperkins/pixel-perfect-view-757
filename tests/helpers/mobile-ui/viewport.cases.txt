async page => {
 const phase='after';
 const runtimeErrors=[];const onError=error=>runtimeErrors.push(String(error));page.on('pageerror',onError);
 const engine=page.context().browser().browserType().name(),base='http://127.0.0.1:4199';
 const artifact='/home/ubuntu/pixel-perfect-view-757/output/playwright/mobile-viewport-2026-10-14/';
 page.setDefaultTimeout(10000);
 const results={engine,version:page.context().browser().version(),phase,cases:[],menus:[],overlays:[],fonts:[]};
 const measure=()=>{
  const root=document.documentElement;
  const box=el=>{if(!el)return null;const r=el.getBoundingClientRect(),s=getComputedStyle(el);return {width:r.width,height:r.height,top:r.top,bottom:r.bottom,left:r.left,right:r.right,minHeight:s.minHeight,maxHeight:s.maxHeight,boxSizing:s.boxSizing,overflowX:s.overflowX,overflowY:s.overflowY,position:s.position}};
  return {clientWidth:root.clientWidth,scrollWidth:root.scrollWidth,scrollHeight:root.scrollHeight,viewportHeight:innerHeight,scroller:document.scrollingElement.tagName,root:box(root),body:box(document.body),shell:box(document.querySelector('.pattern-shell')),card:box(document.querySelector('.card')),header:box(document.querySelector('header')),nav:box(document.querySelector('.mobile-journey')),main:box(document.querySelector('main')),horizontalOutliers:[...document.querySelectorAll('body *')].filter(el=>{const r=el.getBoundingClientRect();return r.width&&r.right>root.clientWidth+1&&getComputedStyle(el).visibility!=='hidden'}).slice(0,12).map(el=>({tag:el.tagName,classes:el.className,box:box(el)})),nestedScrollers:[...document.querySelectorAll('body *')].filter(el=>/(auto|scroll|hidden)/.test(getComputedStyle(el).overflowY)&&el.scrollHeight>el.clientHeight+1).map(el=>({tag:el.tagName,classes:el.className,clientHeight:el.clientHeight,scrollHeight:el.scrollHeight})),meta:document.querySelector('meta[name=viewport]')?.content};
 };

   if(width===320&&path==='/__audit/server-error') await page.screenshot({path:artifact+engine+'-server-error-320-'+phase+'.png',fullPage:true});
  }
 }

  if(font===32&&width===667)await page.screenshot({path:artifact+engine+'-menu-landscape-text-'+phase+'.png'});
 }
 for(const [width,height,font] of [[320,568,16],[667,375,16],[320,568,32],[667,375,32]]){

  const dialog=await page.getByRole('alertdialog').evaluate(el=>{const r=el.getBoundingClientRect();return{left:r.left,right:r.right,top:r.top,bottom:r.bottom,clientHeight:el.clientHeight,scrollHeight:el.scrollHeight,maxHeight:getComputedStyle(el).maxHeight}});

 }

   await page.evaluate(font=>document.documentElement.style.fontSize=font+'px',font);await page.waitForTimeout(100);
   results.fonts.push({width,height,rootFont:font,...await page.evaluate(()=>({heading:getComputedStyle(document.querySelector('.travel-intro h1')).fontSize,date:getComputedStyle(document.querySelector('.travel-date-display time')).fontSize,clientWidth:document.documentElement.clientWidth,scrollWidth:document.documentElement.scrollWidth}))});
  }
 }

}
