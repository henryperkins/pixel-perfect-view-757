async (page) => {
  const origin = new URL(page.url()).origin;
  if (origin !== 'http://127.0.0.1:4276') throw new Error('Run against the local synthetic route fixture on port 4276');
  const browser = page.context().browser();
  const results = [];
  const assert = (value, message) => { if (!value) throw new Error(message); };
  const artifacts = 'output/playwright/upcoming-recovery-20261021';
  const stage = 'current';
  const available = (body = 'neptune', point = 'midheaven', extras = {}) => ({
    needsOnboarding: false,
    day: '2030-01-01', timezone: 'UTC', from: '2030-01-01T12:00:00.000Z', to: '2030-04-01T12:00:00.000Z', computedAt: '2030-01-01T12:01:00.000Z',
    calculation: {
      status: 'available', calculationStatus: 'partial', source: 'Swiss Ephemeris', sourceId: 'synthetic-source',
      from: '2030-01-01T12:00:00.000Z', to: '2030-04-01T12:00:00.000Z', totalMatched: 1, truncated: false, incompleteWindows: 1, missingTargets: [],
      windows: [{ id: 'cycle:synthetic-' + body + '-' + point, transitBody: body, natalPoint: point, aspect: 'conjunction', branch: 'plus', start: '2030-01-10T12:00:00.000Z', end: null, startRange: ['2030-01-10T11:00:00.000Z', '2030-01-10T13:00:00.000Z'], endRange: null, exactAt: ['2030-03-01T12:00:00.000Z'], exactRanges: [['2030-03-01T11:00:00.000Z', '2030-03-01T13:00:00.000Z']], phase: 'upcoming', duration: 'unknown', durationDays: null, confidence: 'incomplete', complete: false, sourceId: 'synthetic-source' }],
      ...extras,
    },
  });
  const unknown = { needsOnboarding: false, timezone: 'UTC', calculation: { status: 'unavailable', reason: 'unknown_time' } };
  const unavailable = { needsOnboarding: false, timezone: 'UTC', calculation: { status: 'unavailable', reason: 'transport' } };
  const long = available('saturn', 'mercury', { calculationStatus: 'complete', incompleteWindows: 0 });
  long.to = long.calculation.to = '2031-01-01T12:00:00.000Z';
  Object.assign(long.calculation.windows[0], { end: '2030-03-10T12:00:00.000Z', endRange: ['2030-03-10T11:00:00.000Z', '2030-03-10T13:00:00.000Z'], duration: 'long', durationDays: 59, complete: true, confidence: 'range' });
  const run = async (name, size, task, initial = {}, clock = false) => {
    const context = await browser.newContext({ viewport: size, isMobile: size.width < 700, hasTouch: size.width < 700, reducedMotion: 'reduce', timezoneId: 'UTC' });
    await context.addInitScript(value => { window.interfaceFixture = value; }, initial);
    const p = await context.newPage();
    if (clock) await p.clock.install();
    p.setDefaultTimeout(7000);
    const errors = [];
    const consoleMessages = [];
    p.on('pageerror', e => errors.push(e.message));
    p.on('console', message => {
      if (['warning', 'error'].includes(message.type()))
        consoleMessages.push({ type: message.type(), text: message.text(), url: message.location().url });
    });
    try {
      await p.goto(origin + '/timing');
      await p.getByRole('button', { name: 'Upcoming', exact: true }).waitFor();
      assert(await p.evaluate(() => Boolean(window.mobileAudit)), 'Synthetic services not loaded');
      assert(await p.title() === 'Pattern mobile fixture' && await p.getByRole('heading', { name: 'Timing', exact: true }).isVisible(), 'Wrong fixture page identity');
      await p.locator('body').ariaSnapshot();
      await p.getByRole('button', { name: 'Upcoming', exact: true }).click();
      const region = p.locator('section[aria-labelledby="upcoming-title"]');
      const details = await task(p, region);
      assert(errors.length === 0, 'Browser runtime errors: ' + errors.join('; '));
      assert(await p.locator('vite-error-overlay').count() === 0, 'Framework error overlay present');
      results.push({ name, size, passed: true, details, errors, consoleMessages });
    } catch (error) {
      results.push({ name, size, passed: false, error: error.message, errors, consoleMessages });
    } finally { await context.close(); }
  };
  const calls = p => p.evaluate(() => window.mobileAudit.state.calls.filter(c => c.name === 'fetchUpcomingCycles'));
  const release = p => p.evaluate(() => window.mobileAudit.releaseNext('fetchUpcomingCycles'));
  const rows = r => r.locator('ul[aria-label="Upcoming calculated windows"] > li');
  const settle = p => p.waitForFunction(() => window.mobileAudit.state.upcomingActive === 0 && window.auditQueryClient.getQueryCache().getAll().filter(q => q.queryKey[0] === 'upcoming-cycles').every(q => q.state.fetchStatus === 'idle'));
  for (const size of [{ width: 1440, height: 1000 }, { width: 390, height: 844 }]) {
    await run('Stalled transport reaches an explicit retry and ignores its late result', size, async (p, r) => {
      await p.waitForFunction(() => window.mobileAudit.state.upcomingActive === 1);
      await p.clock.runFor(60_001);
      const retry = r.getByRole('button', { name: 'Retry calculation', exact: true });
      await retry.waitFor();
      assert(await r.getByRole('alert').isVisible() && await retry.isEnabled(), 'A stalled connection did not reach recovery');
      assert(await p.evaluate(() => window.mobileAudit.state.upcomingSignals[0]?.aborted === true), 'The server function did not receive the transport deadline signal');
      assert((await calls(p)).length === 1, 'The stalled request retried automatically');
      await release(p);
      await settle(p);
      assert(await rows(r).count() === 0 && await r.getByRole('alert').isVisible(), 'A timed-out result replaced the recovery state');
      await p.evaluate(() => { window.mobileAudit.state.hold = []; });
      await retry.focus();
      await p.keyboard.press('Enter');
      await rows(r).first().waitFor();
      await settle(p);
      assert((await calls(p)).length === 2, 'Explicit recovery did not make one new request');
      assert(await r.getByRole('heading', { name: 'Upcoming windows', exact: true }).evaluate(el => document.activeElement === el), 'Timeout recovery lost keyboard focus');
      return { stalledSignalAborted: true, lateResultIgnored: true, attempts: 2, keyboardRecovery: true };
    }, { hold: ['fetchUpcomingCycles'], overrides: { fetchUpcomingCycles: available() } }, true);

    await run('Rapid filters retain only the newest queued selection', size, async (p, r) => {
      await p.waitForFunction(() => window.mobileAudit.state.upcomingActive === 1);
      await r.getByRole('combobox', { name: 'Upcoming area', exact: true }).selectOption('work');
      await r.getByRole('combobox', { name: 'Window duration', exact: true }).selectOption('long');
      await r.getByRole('combobox', { name: 'Date range', exact: true }).selectOption('365');
      await p.waitForFunction(() => window.auditQueryClient.getQueryCache().getAll().some(q => q.queryKey[0] === 'upcoming-cycles' && q.queryKey[4] === 365 && q.queryKey[5] === 'work' && q.queryKey[6] === 'long' && q.state.fetchStatus === 'fetching'));
      const maximum = await p.evaluate(() => window.mobileAudit.state.upcomingMaximumActive);
      assert(maximum === 1, 'Rapid changes overlapped ' + maximum + ' service requests');
      await release(p);
      await p.waitForFunction(() => window.mobileAudit.state.calls.filter(c => c.name === 'fetchUpcomingCycles').length === 2);
      assert(await rows(r).count() === 0, 'Old selection became the newest selection while queued');
      const sent = (await calls(p)).map(c => [c.args.data.horizonDays, c.args.data.domain, c.args.data.duration]);
      assert(JSON.stringify(sent) === JSON.stringify([[90, 'all', 'all'], [365, 'work', 'long']]), 'Obsolete intermediate filter was sent');
      await release(p);
      await r.getByText('Saturn conjunction natal Mercury', { exact: true }).waitFor();
      await settle(p);
      await p.getByRole('button', { name: 'Active', exact: true }).click();
      await p.getByRole('button', { name: 'Upcoming', exact: true }).click();
      assert((await calls(p)).length === 2, 'Revisiting a healthy selection recalculated it');
      const tabOrder = [
        p.getByRole('button', { name: 'Moon & eclipses', exact: true }),
        r.getByRole('combobox', { name: 'Date range', exact: true }),
        r.getByRole('combobox', { name: 'Upcoming area', exact: true }),
        r.getByRole('combobox', { name: 'Window duration', exact: true }),
        r.getByRole('button', { name: 'Clear future filters', exact: true }),
        rows(r).first().locator('summary'),
      ];
      await p.getByRole('button', { name: 'Upcoming', exact: true }).focus();
      for (let i = 0; i < tabOrder.length; i++) {
        await p.keyboard.press('Tab');
        assert(await tabOrder[i].evaluate(el => document.activeElement === el), 'Tab order changed at step ' + i);
      }
      await p.keyboard.press('Enter');
      await rows(r).first().locator('dl').waitFor();
      await p.keyboard.press('Shift+Tab');
      assert(await tabOrder[4].evaluate(el => document.activeElement === el), 'Reverse tab order changed');
      return { maximumActive: maximum, sent, staleResultSuppressed: true, healthyCacheRetained: true, keyboardTabOrder: true };
    }, { hold: ['fetchUpcomingCycles'], upcomingResponses: { 'all/all': available('moon', 'sun'), 'work/all': available('uranus', 'mercury'), 'work/long': long } });

    await run('A returning route shares the active request slot and drops abandoned queued work', size, async (p, r) => {
      await p.waitForFunction(() => window.mobileAudit.state.upcomingActive === 1);
      await r.getByRole('combobox', { name: 'Upcoming area', exact: true }).selectOption('work');
      await p.evaluate(() => window.auditRouter.navigate({ to: '/journal' }));
      await p.getByRole('heading', { name: 'Journal', exact: true }).waitFor();
      await p.evaluate(() => window.auditRouter.navigate({ to: '/timing' }));
      await p.getByRole('button', { name: 'Upcoming', exact: true }).click();
      assert((await calls(p)).length === 1, 'Remount started another request while the abandoned request was active');
      await release(p);
      await p.waitForFunction(() => window.mobileAudit.state.calls.filter(c => c.name === 'fetchUpcomingCycles').length === 2);
      assert((await calls(p)).every(c => c.args.data.domain === 'all'), 'Unmounted queued selection reached the service');
      await release(p);
      await rows(p.locator('section[aria-labelledby="upcoming-title"]')).first().waitFor();
      await settle(p);
      assert(await p.evaluate(() => window.mobileAudit.state.upcomingMaximumActive) === 1, 'Route return lost the shared slot');
      return { requestCount: 2, sharedAcrossRemount: true };
    }, { hold: ['fetchUpcomingCycles'], upcomingResponses: { 'all/all': available(), 'work/all': available('saturn', 'mercury') } });

    await run('Unavailable results recover by one explicit keyboard retry without reload', size, async (p, r) => {
      const retry = r.getByRole('button', { name: 'Retry calculation', exact: true });
      await retry.waitFor();
      await p.screenshot({ path: artifacts + '/' + stage + '-unavailable-' + size.width + '.png' });
      const healthy = available();
      await p.evaluate(result => { window.mobileAudit.state.overrides.fetchUpcomingCycles = result; window.mobileAudit.state.hold = ['fetchUpcomingCycles']; }, healthy);
      await p.locator('body').ariaSnapshot();
      await retry.focus();
      await p.keyboard.press('Enter');
      await p.waitForFunction(() => window.mobileAudit.state.upcomingActive === 1);
      assert(await r.getByRole('heading', { name: 'Upcoming windows', exact: true }).evaluate(el => document.activeElement === el), 'Retry lost keyboard focus');
      assert(await retry.isDisabled(), 'Retry can be repeated while pending');
      assert((await r.getByRole('status').allTextContents()).some(s => s.includes('Calculating')), 'No live loading announcement during retry');
      await release(p);
      await rows(r).first().waitFor();
      await settle(p);
      assert((await calls(p)).length === 2, 'Manual recovery repeated an attempt');
      assert(await r.getByRole('heading', { name: 'Upcoming windows', exact: true }).evaluate(el => document.activeElement === el), 'Successful recovery removed the focus target');
      return { requestCount: 2, keyboardRecovery: true, reloadRequired: false };
    }, { overrides: { fetchUpcomingCycles: unavailable } });

    await run('Another unavailable attempt stays retryable and never retries itself', size, async (p, r) => {
      const retry = r.getByRole('button', { name: 'Retry calculation', exact: true });
      await retry.waitFor();
      await retry.click();
      await p.waitForFunction(() => window.mobileAudit.state.calls.filter(c => c.name === 'fetchUpcomingCycles').length === 2);
      await settle(p);
      assert(await retry.isEnabled(), 'A failed retry became a permanent disabled state');
      assert((await calls(p)).length === 2, 'An unavailable value retried automatically');
      await p.evaluate(result => { window.mobileAudit.state.overrides.fetchUpcomingCycles = result; }, available());
      await retry.click();
      await rows(r).first().waitFor();
      await settle(p);
      assert((await calls(p)).length === 3, 'The next explicit attempt was not independent');
      return { attempts: 3, eachAttemptExplicit: true };
    }, { overrides: { fetchUpcomingCycles: unavailable } });

    await run('Missing target batches retain results and limitations during manual recovery', size, async (p, r) => {
      const retry = r.getByRole('button', { name: 'Retry calculation', exact: true });
      await retry.waitFor();
      await rows(r).first().waitFor();
      await p.evaluate(result => { window.mobileAudit.state.overrides.fetchUpcomingCycles = result; window.mobileAudit.state.hold = ['fetchUpcomingCycles']; }, available());
      await retry.click();
      await p.waitForFunction(() => window.mobileAudit.state.upcomingActive === 1);
      assert(await rows(r).first().isVisible(), 'Retry hid the successful partial calculation');
      assert((await r.getByRole('status').allTextContents()).some(s => s.includes('Moon')), 'Missing target disclosure disappeared before recovery');
      await release(p);
      await settle(p);
      assert(await retry.count() === 0, 'Healthy geometry-partial result still offers failed-batch recovery');
      assert((await r.getByRole('status').allTextContents()).some(s => s.includes('incomplete')), 'Successful recovery erased the remaining bound limitation');
      return { partialResultsRetained: true, limitationsRetained: true };
    }, { overrides: { fetchUpcomingCycles: available('neptune', 'midheaven', { missingTargets: ['moon'] }) } });

    for (const reason of ['transport', 'timeout', 'invalid_response']) {
      await run('Transient unavailable refresh retains previous facts: ' + reason, size, async (p, r) => {
        const retry = r.getByRole('button', { name: 'Retry calculation', exact: true });
        await retry.waitFor();
        await rows(r).first().waitFor();
        await p.evaluate(result => { window.mobileAudit.state.overrides.fetchUpcomingCycles = result; }, { ...unavailable, calculation: { status: 'unavailable', reason } });
        await retry.click();
        await p.waitForFunction(() => window.mobileAudit.state.calls.filter(c => c.name === 'fetchUpcomingCycles').length === 2);
        await settle(p);
        assert(await rows(r).first().isVisible(), 'A transient unavailable response erased the previous calculation');
        assert(await r.getByRole('alert').isVisible(), 'The retained calculation has no failed-refresh notice');
        assert((await r.getByRole('status').allTextContents()).some(s => s.includes('Moon')), 'The retained result lost its missing-target disclosure');
        await p.evaluate(result => { window.mobileAudit.state.overrides.fetchUpcomingCycles = result; }, available());
        await retry.click();
        await p.waitForFunction(() => window.mobileAudit.state.calls.filter(c => c.name === 'fetchUpcomingCycles').length === 3);
        await settle(p);
        await r.getByRole('alert').waitFor({ state: 'detached' });
        await retry.waitFor({ state: 'detached' });
        assert(await r.getByRole('alert').count() === 0 && await retry.count() === 0, 'Recovery left a false unavailable state');
        assert((await calls(p)).length === 3, 'A transient result automatically retried');
        return { reason, previousFactsRetained: true, attempts: 3 };
      }, { overrides: { fetchUpcomingCycles: available('neptune', 'midheaven', { missingTargets: ['moon'] }) } });
    }

    for (const reason of ['unknown_time', 'anchors_unavailable', 'unqualified_anchors', 'not_configured']) {
      await run('Unsupported results replace previous facts rather than retaining them as current: ' + reason, size, async (p, r) => {
      const retry = r.getByRole('button', { name: 'Retry calculation', exact: true });
      await retry.waitFor();
      await rows(r).first().waitFor();
      await p.evaluate(result => { window.mobileAudit.state.overrides.fetchUpcomingCycles = result; }, { ...unknown, calculation: { status: 'unavailable', reason } });
      await retry.click();
      await p.waitForFunction(() => window.mobileAudit.state.calls.filter(c => c.name === 'fetchUpcomingCycles').length === 2);
      await settle(p);
      assert(await rows(r).count() === 0, 'Unsupported state kept old facts as a current result');
      assert(await r.getByRole('alert').count() === 0, 'Unsupported state became a transient transport failure');
      if (reason === 'unknown_time') {
        assert((await r.getByRole('status').allTextContents()).some(s => s.includes('unknown birth time')), 'Unknown-time limitation missing');
        assert(await retry.count() === 0, 'Unknown birth time offered a futile retry');
      }
      assert((await calls(p)).length === 2, 'Unsupported state retried automatically');
      return { reason, previousFactsRemoved: true, unsupportedStateDistinct: true };
    }, { overrides: { fetchUpcomingCycles: available('neptune', 'midheaven', { missingTargets: ['moon'] }) } });
    }

    await run('A failed refresh keeps the previous partial facts and can recover again', size, async (p, r) => {
      const retry = r.getByRole('button', { name: 'Retry calculation', exact: true });
      await retry.waitFor();
      await rows(r).first().waitFor();
      await p.evaluate(() => { window.mobileAudit.state.fail = ['fetchUpcomingCycles']; });
      await retry.click();
      await p.waitForFunction(() => window.mobileAudit.state.calls.filter(c => c.name === 'fetchUpcomingCycles').length === 2);
      await settle(p);
      assert(await rows(r).first().isVisible(), 'A failed refresh hid previously calculated windows');
      assert(await r.getByRole('alert').isVisible(), 'Failed refresh has no failure announcement');
      assert((await r.getByRole('status').allTextContents()).some(s => s.includes('Moon')), 'Failed refresh hid its missing target');
      assert(await retry.count() === 1, 'Failed refresh duplicated the recovery control');
      await p.evaluate(result => { window.mobileAudit.state.fail = []; window.mobileAudit.state.overrides.fetchUpcomingCycles = result; }, available());
      await retry.click();
      await p.waitForFunction(() => window.mobileAudit.state.calls.filter(c => c.name === 'fetchUpcomingCycles').length === 3);
      await settle(p);
      await r.getByRole('alert').waitFor({ state: 'detached' });
      await retry.waitFor({ state: 'detached' });
      assert(await r.getByRole('alert').count() === 0 && await retry.count() === 0, 'Successful recovery left a false failure state');
      assert((await calls(p)).length === 3, 'Failed refresh triggered an automatic retry');
      return { previousFactsRetained: true, failureAnnounced: true, attempts: 3 };
    }, { overrides: { fetchUpcomingCycles: available('neptune', 'midheaven', { missingTargets: ['moon'] }) } });

    await run('Transport errors make one attempt and preserve an explicit recovery action', size, async (p, r) => {
      const retry = r.getByRole('button', { name: 'Retry calculation', exact: true });
      await retry.waitFor({ timeout: 15000 });
      assert((await calls(p)).length === 1, 'A transport error triggered automatic retries');
      await p.evaluate(result => { window.mobileAudit.state.fail = []; window.mobileAudit.state.overrides.fetchUpcomingCycles = result; }, available());
      await retry.click();
      await rows(r).first().waitFor();
      await settle(p);
      assert((await calls(p)).length === 2, 'Explicit recovery did not make exactly one attempt');
      return { initialAttempts: 1, explicitRecovery: true };
    }, { fail: ['fetchUpcomingCycles'] });

    await run('Unknown birth time remains unsupported rather than offering a futile retry', size, async (p, r) => {
      await r.getByRole('status').filter({ hasText: /unknown birth time/ }).waitFor();
      assert((await r.getByRole('status').innerText()).includes('unknown birth time'), 'Unknown-time limitation missing');
      assert(await r.getByRole('button', { name: 'Retry calculation', exact: true }).count() === 0, 'Unsupported result offers service recovery');
      await p.getByRole('button', { name: 'Active', exact: true }).click();
      await p.getByRole('button', { name: 'Upcoming', exact: true }).click();
      assert((await calls(p)).length === 1, 'Unsupported result automatically retried');
      return { unknownTimeDistinct: true, retained: true };
    }, { overrides: { fetchUpcomingCycles: unknown } });
  }
  for (const size of [{ width: 1440, height: 1000 }, { width: 390, height: 844 }, { width: 320, height: 844 }]) {
    await run('Long titles and calculated details reflow at 200 percent text', size, async (p, r) => {
      await rows(r).first().waitFor();
      await p.addStyleTag({ content: 'html { font-size: 200% !important; }' });
      const row = rows(r).first(), summary = row.locator('summary');
      await summary.focus();
      await p.keyboard.press('Enter');
      await row.locator('dl').waitFor();
      await p.screenshot({ path: artifacts + '/' + stage + '-text200-' + size.width + '.png' });
      await row.screenshot({ path: artifacts + '/' + stage + '-facts200-' + size.width + '.png' });
      const metrics = await summary.evaluate(el => {
        const range = document.createRange(); range.selectNodeContents(el);
        return { textMaxRight: Math.max(...[...range.getClientRects()].map(r => r.right)), summaryRight: el.getBoundingClientRect().right, documentWidth: document.documentElement.scrollWidth, innerWidth, fontSize: getComputedStyle(el).fontSize };
      });
      assert(metrics.documentWidth <= size.width + 1 && metrics.innerWidth <= size.width + 1, 'Layout viewport expanded beyond configured width: ' + JSON.stringify(metrics));
      assert(metrics.textMaxRight <= metrics.summaryRight + 1, 'Title text exceeded its disclosure');
      for (const control of await r.getByRole('combobox').all()) {
        const box = await control.boundingBox();
        assert(box.x >= 0 && box.x + box.width <= size.width + 1 && box.height >= 44, 'Select clipped or too small');
      }
      assert((await row.innerText()).includes('Outside the calculated range'), 'Unknown bound lost');
      assert((await row.innerText()).includes('Unknown'), 'Unknown duration lost');
      assert((await row.innerText()).includes('timing varies'), 'Birth-time uncertainty lost');
      assert((await row.locator('dt').allTextContents()).length === 5, 'Calculated facts lost');
      const record = r.locator('summary').filter({ hasText: /^Calculation record$/ });
      await record.focus(); await p.keyboard.press('Enter');
      await record.locator('..').locator('p').waitFor();
      assert((await record.locator('..').innerText()).includes('Swiss Ephemeris'), 'Source record inaccessible');
      return { ...metrics, keyboardDisclosures: true, configuredWidth: size.width };
    }, { overrides: { fetchUpcomingCycles: available() } });
  }
  await run('Retry and filters remain reachable on a narrow phone with enlarged text', { width: 320, height: 844 }, async (p, r) => {
    const retry = r.getByRole('button', { name: 'Retry calculation', exact: true });
    await retry.waitFor();
    await p.addStyleTag({ content: 'html { font-size: 200% !important; }' });
    await retry.focus();
    const controls = [...await r.getByRole('combobox').all(), ...await r.getByRole('button').all()];
    for (const control of controls) {
      const box = await control.boundingBox();
      assert(box.x >= 0 && box.x + box.width <= 321 && box.height >= 44, 'An enlarged control overflows or has a small touch target');
    }
    assert(await p.evaluate(() => document.documentElement.scrollWidth <= 321 && innerWidth <= 321), 'Unavailable state expands the narrow viewport');
    await p.screenshot({ path: artifacts + '/' + stage + '-retry200-320.png' });
    await p.evaluate(result => { window.mobileAudit.state.overrides.fetchUpcomingCycles = result; }, available());
    await p.keyboard.press('Enter');
    await rows(r).first().waitFor();
    assert(await r.getByRole('heading', { name: 'Upcoming windows', exact: true }).evaluate(el => document.activeElement === el), 'Enlarged keyboard recovery lost focus');
    return { keyboardRecovery: true, configuredWidth: 320, enlargedTextPercent: 200, controls: controls.length };
  }, { overrides: { fetchUpcomingCycles: unavailable } });
  return { stage, results, passed: results.filter(r => r.passed).length, failed: results.filter(r => !r.passed).length };
}
