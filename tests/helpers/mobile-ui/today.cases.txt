async (page) => {
  const browser = page.context().browser();
  const origin = new URL(page.url()).origin;
  const checks = [];
  const assert = (condition, message, details) => {
    if (!condition) throw new Error(`${message}: ${JSON.stringify(details)}`);
  };
  const click = async (p, locator) => {
    await p.locator('body').ariaSnapshot();
    await locator.evaluate((element) => element.scrollIntoView({ block: 'center', behavior: 'instant' }));
    await locator.tap();
  };
  const run = async (name, task) => {
    const context = await browser.newContext({
      viewport: { width: 390, height: 844 },
      isMobile: true,
      hasTouch: true,
      reducedMotion: 'reduce',
    });
    const p = await context.newPage();
    p.setDefaultTimeout(10000);
    try {
      await p.goto(origin + '/');
      await p.getByRole('textbox', { name: 'Reflection note' }).waitFor();
      assert(await p.evaluate(() => window.mobileAudit?.state.ownerId === 'mobile-audit-member'), 'Synthetic fixture required');
      await p.evaluate(() => document.fonts.ready);
      checks.push({ name, passed: true, details: await task(p) });
    } catch (error) {
      checks.push({ name, passed: false, error: String(error) });
    } finally {
      await context.close();
    }
  };

  await run('F1: enlarged check-in choices remain distinct and each value can be selected', async (p) => {
    await p.evaluate(async () => {
      window.mobileAudit.state.wellness.sleepEnabled = true;
      window.mobileAudit.state.wellness.stressEnabled = true;
      await window.auditQueryClient.invalidateQueries({ queryKey: ['today'] });
    });
    await p.getByRole('group', { name: 'Stress', exact: true }).waitFor();
    const samples = [];
    for (const [width, height, font] of [[320, 568, 16], [360, 800, 16], [390, 844, 16], [440, 956, 16], [667, 375, 16], [844, 390, 16], [320, 568, 32], [390, 844, 32]]) {
      await p.setViewportSize({ width, height });
      await p.evaluate((size) => document.documentElement.style.fontSize = size + 'px', font);
      // Let Shell's measured header and navigation settle before touch coordinates are chosen.
      await p.evaluate(() => new Promise((resolve) => requestAnimationFrame(() => requestAnimationFrame(() => requestAnimationFrame(resolve)))));
      const geometry = await p.evaluate(() => [...document.querySelectorAll('.today-check-in-options')].map((row) => {
        const cells = [...row.querySelectorAll('label')].map((label) => {
          const cell = label.getBoundingClientRect(), paint = label.querySelector('span').getBoundingClientRect();
          return { left: paint.left, right: paint.right, cellLeft: cell.left, cellRight: cell.right, width: cell.width, height: cell.height };
        });
        return { cells, fits: cells.every((cell) => cell.left >= cell.cellLeft - 1 && cell.right <= cell.cellRight + 1 && cell.width >= 24 && cell.height >= 24) };
      }));
      assert(geometry.length === 4 && geometry.every((row) => row.fits), 'A choice overlaps another cell or loses its minimum target', { width, height, font, geometry });
      if (font === 32) {
        for (const label of ['Mood', 'Energy', 'Sleep', 'Stress']) {
          for (let value = 1; value <= 5; value++) {
            const radio = p.getByRole('radio', { name: new RegExp('^' + label + ': ' + value + ' of 5,') });
            await click(p, radio.locator('..'));
            await p.waitForFunction((name) => [...document.querySelectorAll('input[type=radio]')].some((input) => input.getAttribute('aria-label') === name && input.checked), await radio.getAttribute('aria-label'));
            assert(await radio.isChecked(), 'Tapping a choice selected the wrong value', { width, label, value });
          }
        }
      }
      samples.push({ width, height, font, geometry });
    }
    return samples;
  });

  await run('F2: saved-value review reflows at narrow widths and enlarged text', async (p) => {
    const note = p.getByRole('textbox', { name: 'Reflection note' });
    await note.fill('A synthetic draft whose save will fail before commit.');
    await p.evaluate(() => { window.mobileAudit.state.fail = ['saveDailyCheckIn']; });
    await click(p, p.getByRole('button', { name: 'Save check-in', exact: true }));
    await p.getByRole('button', { name: 'Check save', exact: true }).waitFor();
    await click(p, p.getByRole('button', { name: 'Check save', exact: true }));
    const review = p.getByRole('button', { name: 'Keep my draft and review before saving', exact: true });
    await review.waitFor();
    const samples = [];
    for (const [width, height, font] of [[320, 568, 16], [360, 800, 16], [320, 568, 32], [390, 844, 32]]) {
      await p.setViewportSize({ width, height });
      await p.evaluate((size) => document.documentElement.style.fontSize = size + 'px', font);
      const geometry = await review.evaluate((button) => {
        const box = button.getBoundingClientRect(), parent = button.parentElement.getBoundingClientRect();
        const range = document.createRange();
        range.selectNodeContents(button);
        const text = range.getBoundingClientRect();
        return { client: document.documentElement.clientWidth, scroll: document.documentElement.scrollWidth, fitsParent: box.left >= parent.left - 1 && box.right <= parent.right + 1, fitsText: text.top >= box.top - 1 && text.bottom <= box.bottom + 1 };
      });
      assert(geometry.scroll <= geometry.client && geometry.fitsParent && geometry.fitsText, 'The review action overflows or clips its label', { width, height, font, geometry });
      samples.push({ width, height, font, geometry });
    }
    await p.evaluate(() => { document.documentElement.style.fontSize = ''; window.mobileAudit.state.fail = []; });
    await click(p, review);
    await click(p, p.getByRole('button', { name: 'Save check-in', exact: true }));
    await p.waitForFunction(() => document.querySelector('.today-check-in-status').textContent.startsWith('Saved '));
    assert(await note.inputValue() === 'A synthetic draft whose save will fail before commit.', 'Review discarded the draft');

  });

  await run('F3: Sources labels remain inside the button when opening and closing with enlarged text', async (p) => {
    const toggle = p.getByRole('button', { name: /sources behind this reading/i });

    for (const [width, height, font] of [[320, 568, 16], [320, 568, 32], [390, 844, 32], [667, 375, 32]]) {
      await p.setViewportSize({ width, height });
      await p.evaluate((size) => document.documentElement.style.fontSize = size + 'px', font);
      for (const open of [false, true]) {

          const box = button.getBoundingClientRect(), range = document.createRange();
          range.selectNodeContents(button);
          const text = range.getBoundingClientRect();
          return { buttonHeight: box.height, textHeight: text.height, containsText: text.top >= box.top - 1 && text.bottom <= box.bottom + 1 && text.left >= box.left - 1 && text.right <= box.right + 1 };
        });
        assert(geometry.containsText, 'The Sources label extends outside its hit area', { width, height, font, open, geometry });
        assert(await toggle.getAttribute('aria-expanded') === String(open), 'Sources has the wrong disclosure state', { open });
        samples.push({ width, height, font, open, geometry });
        await click(p, toggle);
      }

  });

  await run('F4: an uncertain save exposes a nearby action and reconciles without losing newer edits', async (p) => {

    assert(await actions.getByRole('button', { name: 'Try saving again', exact: true }).count() === 0, 'A disabled retry competes with the recovery action');
    const geometry = await checkSave.evaluate((button) => {
      const action = button.getBoundingClientRect(), status = document.querySelector('.today-check-in-status').getBoundingClientRect();
      return { verticalGap: Math.max(0, action.top - status.bottom, status.top - action.bottom) };

    assert(geometry.verticalGap <= 24, 'Recovery is separated from its status', geometry);
    await p.evaluate(() => { window.mobileAudit.state.delay = 0; window.mobileAudit.state.fail = ['fetchToday']; });
    await click(p, checkSave);

    await click(p, checkSave);
    await p.getByRole('button', { name: 'Save check-in', exact: true }).waitFor();
    assert(await note.inputValue() === 'Newer synthetic reflection typed while the response was lost.', 'Confirmation replaced newer edits');
    assert(await p.getByRole('button', { name: 'Save check-in', exact: true }).isEnabled(), 'Newer edits cannot be saved after confirmation');
    await p.getByText('Draft saved on this device', { exact: true }).waitFor();
    assert(await p.getByText('Draft saved on this device', { exact: true }).count() === 1, 'Newer edits lost their device checkpoint');
    const writes = await p.evaluate(() => window.mobileAudit.state.calls.filter((call) => call.name === 'saveDailyCheckIn').length);
    assert(writes === 1, 'Checking confirmation issued another save', { writes });
    return { ...geometry, writes };
  });
  await run('M1: an offline save sends its original snapshot and retains later edits', async (p) => {
    const note = p.getByRole('textbox', { name: 'Reflection note' });

    assert(writes.length === 1 && writes[0] === original, 'Reconnection changed the submitted values', writes);
    assert(await note.inputValue() === later, 'The queued acknowledgment consumed newer edits');

    assert(await p.getByRole('button', { name: 'Save check-in', exact: true }).isEnabled(), 'Later edits cannot be saved after recovery');

  });

  await run('M2: all Today recovery actions reflow at narrow and enlarged text sizes', async (p) => {
    const samples = [];

      await action.waitFor();
      for (const [width, height, font] of [[320, 568, 16], [390, 844, 16], [320, 568, 32], [390, 844, 32], [667, 375, 32]]) {

        // than expecting resize to pull an already-focused control into view.
        await p.evaluate(() => { if (document.activeElement instanceof HTMLElement) document.activeElement.blur(); });
        await action.focus();
        await p.evaluate(() => new Promise((resolve) => requestAnimationFrame(() => requestAnimationFrame(() => requestAnimationFrame(resolve)))));
        const geometry = await action.evaluate((element) => {
          const box = element.getBoundingClientRect(), parent = element.parentElement.getBoundingClientRect(), range = document.createRange();
          range.selectNodeContents(element);
          const text = range.getBoundingClientRect();
          const header = document.querySelector('header'), nav = document.querySelector('.mobile-journey');
          const viewport = window.visualViewport;
          const clearTop = Math.max(viewport?.offsetTop ?? 0, getComputedStyle(header).position === 'sticky' ? header.getBoundingClientRect().bottom : 0);
          const clearBottom = Math.min((viewport?.offsetTop ?? 0) + (viewport?.height ?? innerHeight), getComputedStyle(nav).position === 'fixed' ? nav.getBoundingClientRect().top : Infinity);
          return { client: document.documentElement.clientWidth, scroll: document.documentElement.scrollWidth, fitsParent: box.left >= parent.left - 1 && box.right <= parent.right + 1, containsText: text.top >= box.top - 1 && text.bottom <= box.bottom + 1, clearFocus: box.top >= clearTop - 1 && box.bottom <= clearBottom + 1 };
        });
        assert(geometry.scroll <= geometry.client && geometry.fitsParent && geometry.containsText && geometry.clearFocus, 'A recovery action or its label overflows or is covered', { state, width, height, font, geometry });
        samples.push({ state, width, height, font, geometry });
      }
      await p.setViewportSize({ width: 390, height: 844 });
      await p.evaluate(() => document.documentElement.style.fontSize = '');

    assert(await note.inputValue() === 'Replacement draft requires its own Save.', 'Reconnection replaced the new draft');
    await click(p, p.getByRole('button', { name: 'Try saving again', exact: true }));
    await p.waitForFunction(() => document.querySelector('.today-check-in-status').textContent.startsWith('Saved '));
    const writes = await p.evaluate(() => window.mobileAudit.state.calls.filter((c) => c.name === 'saveDailyCheckIn').map((c) => c.args.data.note));
    assert(writes.length === 1 && writes[0] === 'Replacement draft requires its own Save.', 'The replacement did not retain its independent save intent', writes);
    return { writes };
  });

  await run('M5: Save and Check save retain 48px primary targets', async (p) => {
    const action = p.locator('.today-check-in-save button').first(), samples = [];
    for (const [width, height] of [[320, 568], [360, 800], [390, 844], [440, 956], [667, 375], [844, 390]]) {
      await p.setViewportSize({ width, height });
      const box = await action.boundingBox();
      assert(box.width >= 48 && box.height >= 48, 'Save loses its primary target size', { width, height, box });
      samples.push({ width, height, box });
    }
    await p.getByRole('textbox', { name: 'Reflection note' }).fill('Synthetic recovery target check.');
    await p.evaluate(() => { window.mobileAudit.state.fail = ['saveDailyCheckIn']; });
    await click(p, p.getByRole('button', { name: 'Save check-in', exact: true }));
    await p.getByRole('button', { name: 'Check save', exact: true }).waitFor();
    // Measure the resting target after the browser releases touch :active feedback.
    await p.waitForFunction(() => !document.querySelector('.today-check-in-save button:active'));
    const recovery = await action.boundingBox();
    assert(recovery.width >= 48 && recovery.height >= 48, 'Check save loses its primary target size', recovery);
    return { samples, recovery };
  });

  await run('M6: saved-value review returns keyboard focus to Save', async (p) => {
    await p.getByRole('textbox', { name: 'Reflection note' }).fill('Synthetic keyboard review note.');
    await p.evaluate(() => { window.mobileAudit.state.fail = ['saveDailyCheckIn']; });
    await click(p, p.getByRole('button', { name: 'Save check-in', exact: true }));
    await p.getByRole('button', { name: 'Check save', exact: true }).waitFor();
    await click(p, p.getByRole('button', { name: 'Check save', exact: true }));
    const review = p.getByRole('button', { name: 'Keep my draft and review before saving', exact: true });
    await review.waitFor();
    await p.locator('body').ariaSnapshot();
    await review.focus();
    await p.keyboard.press('Enter');
    await p.waitForFunction(() => document.activeElement?.matches('.today-check-in-save button[type="submit"]'));
    assert(await p.getByRole('button', { name: 'Save check-in', exact: true }).isEnabled(), 'Focus returned to an unavailable action');
    return { active: await p.evaluate(() => document.activeElement.textContent) };
  });
  return { browser: browser.version(), checks };
