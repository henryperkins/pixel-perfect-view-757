async page => {

  });
  for(const viewport of [{width:360,height:800},{width:667,height:375}])for(const field of [
    {name:'Today reflection',path:'/',ready:'.today-reading',selector:'.today-check-in-note textarea'},

    {name:'Journal edit',path:'/journal',ready:'.journal-entry',selector:'.journal-entry-editor textarea',edit:true},

    await loc.evaluate(el=>el.scrollTop=0);await settle();const top=await state();await page.mouse.wheel(0,-400);await page.waitForTimeout(220);const topAfter=await state();
    check(Math.abs(topAfter.documentY-top.documentY)<2,'Top boundary chained to page');
    await page.mouse.move(5,viewport.height/2);await page.mouse.wheel(0,200);await page.waitForTimeout(220);const outside=await state();
    check(outside.documentY>topAfter.documentY,'Document cannot scroll outside the field');check(await loc.inputValue()===value,'Field input changed');
    return{viewport,start,interior,bottom,bottomAfter,top,topAfter,outside};
  });
  const viewports=[{width:320,height:568,font:16},{width:360,height:800,font:16},{width:440,height:956,font:16},{width:667,height:375,font:16},{width:844,height:390,font:16},{width:1280,height:800,font:16},{width:320,height:568,font:32},{width:844,height:390,font:32}];
  for(const viewport of viewports)await run('G03 all six filters by taps '+viewport.width+' font'+viewport.font,async()=>{
    await page.setViewportSize({width:viewport.width,height:viewport.height});await go('/timing','.timing-area-options');await page.evaluate(font=>document.documentElement.style.fontSize=font+'px',viewport.font);await settle();

    const positions=[0];let keyboardScroll=0,buttonGeometry=[];

      await more.waitFor();check(await previous.isDisabled(),'Previous not disabled at start');check(await more.getAttribute('aria-controls')===await region.getAttribute('id'),'Buttons do not control this region');

      await region.evaluate(el=>el.scrollLeft=0);await center(page.locator('.pattern-placement-controls'));
    }else check(await more.count()===0,'Controls remain visible without overflow');

  });

  result.root=await page.evaluate(()=>({scrollingElement:document.scrollingElement.tagName,overscroll:getComputedStyle(document.documentElement).overscrollBehavior,touchAction:getComputedStyle(document.documentElement).touchAction}));
