async page=>{
 const engine=page.context().browser().browserType().name(),result={engine,version:page.context().browser().version(),evidence:'Render accessibility tree, not VoiceOver/TalkBack',cases:[],pageErrors:[]};
 const base='http://127.0.0.1:4199',out='/home/ubuntu/pixel-perfect-view-757/output/playwright/mobile-overlays-2026-10-14/';
 page.setDefaultTimeout(7000);page.on('pageerror',e=>result.pageErrors.push(e.message));await page.emulateMedia({reducedMotion:'reduce'});
 const settle=()=>page.evaluate(()=>new Promise(r=>requestAnimationFrame(()=>requestAnimationFrame(r))));
 const check=(condition,message)=>{if(!condition)throw Error(message)};
 await page.goto(base+'/pattern');await page.locator('#pattern-placements').waitFor();
 await page.evaluate(()=>{const pattern=structuredClone(window.mobileAudit.pattern);pattern.placements=Array.from({length:12},(_,i)=>({key:'body-'+i,label:['Sun','Moon','Mercury','Venus','Mars','Jupiter','Saturn','Uranus','Neptune','Pluto','North node','Chiron'][i],position:['Aries 5° 12′','Taurus 15° 04′','Gemini 20° 33′'][i%3],house:i+1,retrograde:i%3===0}));window.auditQueryClient.setQueryData(['pattern',window.mobileAudit.state.ownerId],pattern)});
 const summary=page.locator('#pattern-placements > summary');await summary.evaluate(el=>el.scrollIntoView({block:'center',behavior:'instant'}));await summary.click();
 for(const test of [{width:320,height:568,font:16},{width:360,height:800,font:16},{width:440,height:956,font:16},{width:667,height:375,font:16},{width:844,height:390,font:16},{width:1280,height:800,font:16},{width:320,height:568,font:32}]){
  try{
   await page.setViewportSize({width:test.width,height:test.height});await page.evaluate(font=>document.documentElement.style.fontSize=font+'px',test.font);await page.evaluate(()=>document.fonts.ready);await settle();
   const region=page.getByRole('region',{name:'Scrollable natal placements'});await region.evaluate(el=>{el.scrollIntoView({block:'center',behavior:'instant'});el.scrollLeft=0});await settle();
   const aria=await page.locator('#pattern-placements').ariaSnapshot();check(aria.includes('table "Calculated natal placements"'),'Native table missing');check(aria.includes('columnheader "Sign & degree"'),'Column header missing');check(aria.includes('rowheader "Sun"'),'Row header missing');
   const before=await region.evaluate(el=>({rootWidth:document.documentElement.scrollWidth,client:el.clientWidth,scroll:el.scrollWidth,tableDisplay:getComputedStyle(el.querySelector('table')).display,rowDisplay:getComputedStyle(el.querySelector('tr')).display,firstLeft:el.querySelector('tbody th').getBoundingClientRect().left,position:getComputedStyle(el.querySelector('tbody th')).position}));
   check(before.rootWidth===test.width,'Table causes document overflow');check(before.tableDisplay==='table'&&before.rowDisplay==='table-row','Native table layout replaced');
   await region.focus();await page.keyboard.press('ArrowRight');await page.waitForTimeout(180);const keyboardScroll=await region.evaluate(el=>el.scrollLeft);if(before.scroll>before.client)check(keyboardScroll>0,'ArrowRight cannot scroll region');
   await region.evaluate(el=>el.scrollLeft=el.scrollWidth);await settle();const after=await region.evaluate(el=>({left:el.scrollLeft,firstLeft:el.querySelector('tbody th').getBoundingClientRect().left}));check(Math.abs(after.firstLeft-before.firstLeft)<2,'Row header scrolls out of view');
   await page.screenshot({path:out+engine+'-placements-'+test.width+'x'+test.height+'-font'+test.font+'.png',animations:'disabled'});
   result.cases.push({test,pass:true,before,after,keyboardScroll,aria});
  }catch(e){result.cases.push({test,pass:false,error:String(e)})}
 }
 return result;
}
